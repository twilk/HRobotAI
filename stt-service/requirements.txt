# stt-service runtime deps (single source of dep truth; the Dockerfile installs this).
# DISTINCT image from agent-service on purpose (PoC §1): faster-whisper needs CTranslate2, NOT
# torch/SB3, and mixing them would bloat an already-heavy RL image for no benefit.
#
# Base is python:3.12-slim, matching agent-service — CTranslate2 publishes 3.12 wheels, so there is
# no reason to diverge from the fleet's Python.

# --- API scaffold (pins kept in lockstep with agent-service where shared) ---
fastapi==0.115.6
uvicorn[standard]==0.34.0
pydantic==2.10.4
httpx==0.27.2
# Multipart audio upload (UploadFile) — FastAPI requires this for File(...) parameters.
python-multipart==0.0.20
# Keycloak bearer verification for /voice/* (RS256 against realm JWKS); [cryptography] pulls the
# backend python-jose needs for RSA signature checks. See app/deps.py.
python-jose[cryptography]==3.3.0

# --- local STT engine (RODO: audio never leaves this container) ---
# faster-whisper pulls ctranslate2 (inference), tokenizers, onnxruntime (VAD) and av (audio decode).
# The `small` PL model (~490 MB) is fetched once from HuggingFace at first use and cached in
# STT_MODEL_CACHE, after which the service runs fully offline.
faster-whisper==1.2.1

# --- in-container test runner (smoke evidence runs pytest inside the image) ---
pytest==8.3.4
